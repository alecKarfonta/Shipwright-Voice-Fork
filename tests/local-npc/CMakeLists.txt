cmake_minimum_required(VERSION 3.20)
project(LocalNpcTests LANGUAGES CXX)
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(npc_client_test npc_client_test.cpp ../../soh/soh/Enhancements/local-npc/NpcClient.cpp)
target_compile_features(npc_client_test PRIVATE cxx_std_17)
target_compile_definitions(npc_client_test PRIVATE SOH_LOCAL_NPC=1)
target_include_directories(npc_client_test PRIVATE ../../soh/soh/Enhancements/local-npc)
target_link_libraries(npc_client_test PRIVATE CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
enable_testing()
add_executable(dialogue_text_test dialogue_text_test.cpp)
target_compile_features(dialogue_text_test PRIVATE cxx_std_17)
target_include_directories(dialogue_text_test PRIVATE ../../soh/soh/Enhancements/local-npc)
add_test(NAME dialogue_text COMMAND dialogue_text_test)
add_executable(voice_pack_test voice_pack_test.cpp)
target_compile_features(voice_pack_test PRIVATE cxx_std_17)
target_include_directories(voice_pack_test PRIVATE ../../soh/soh/Enhancements/local-npc)
target_link_libraries(voice_pack_test PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME voice_pack COMMAND voice_pack_test)
add_test(NAME npc_client COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/fixture.py $<TARGET_FILE:npc_client_test>)
set_tests_properties(npc_client PROPERTIES TIMEOUT 30)
