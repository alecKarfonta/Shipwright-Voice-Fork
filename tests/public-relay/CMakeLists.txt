cmake_minimum_required(VERSION 3.26)
project(relay_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(USE_TLS ON)
set(IXWEBSOCKET_INSTALL OFF)
include(FetchContent)
FetchContent_Declare(ixwebsocket URL https://codeload.github.com/machinezone/IXWebSocket/tar.gz/refs/tags/v11.4.6 URL_HASH SHA256=c024334f8e45980836c67008979a884d6dcc5ef067dd2eb1fa7241f4c17ddc32)
FetchContent_MakeAvailable(ixwebsocket)
find_package(nlohmann_json REQUIRED)
add_executable(relay_probe probe.cpp ../../soh/soh/Network/RelayWebSocket.cpp)
target_compile_definitions(relay_probe PRIVATE SOH_PUBLIC_RELAY=1)
target_include_directories(relay_probe PRIVATE ../../soh/soh/Network)
target_link_libraries(relay_probe PRIVATE ixwebsocket nlohmann_json::nlohmann_json)

enable_testing()
add_executable(browser_model_test browser_model_test.cpp)
target_compile_options(browser_model_test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-UNDEBUG>)
add_test(NAME browser_model COMMAND browser_model_test)

add_executable(fellowship_model_test fellowship_model_test.cpp)
target_link_libraries(fellowship_model_test PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(fellowship_model_test PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-UNDEBUG>)
add_test(NAME fellowship_model COMMAND fellowship_model_test)
